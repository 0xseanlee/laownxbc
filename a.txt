def decode_adrp(word, pc):
    if (word & 0x9F000000) != 0x90000000:
        return None

    rd = word & 0x1F

    immlo = (word >> 29) & 0x3
    immhi = (word >> 5) & 0x7FFFF

    imm21 = (immhi << 2) | immlo
    imm21 = s64(imm21, 21)

    target_page = (pc & ~0xFFF) + (imm21 << 12)

    return rd, target_page

def decode_ldr64_unsigned(word):
    if (word & 0xFFC00000) != 0xF9400000:
        return None

    rt = word & 0x1F
    rn = (word >> 5) & 0x1F
    imm12 = (word >> 10) & 0xFFF

    return rt, rn, imm12 * 8
